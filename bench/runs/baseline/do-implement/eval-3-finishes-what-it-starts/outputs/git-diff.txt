+/**
+ * Persist the same status snapshot `status --json` prints to <neutral-state>/export.json, so it
+ * can be diffed, archived, or shipped to another system without re-running the CLI. Destination

+function cmdExport(o) {

+  const scope = scopeOf(o);
+  const root = ctx.registry && ctx.registry.stateRoot
+    ? ctx.registry.stateRoot
